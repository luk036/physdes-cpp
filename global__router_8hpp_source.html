92 void refresh_bbox() {
93 auto bbox = pt.hull_with(pt);
94 for (auto* child : children) {
95 bbox = bbox.hull_with(child->subtree_bbox);
96 }
97 subtree_bbox = bbox;

101 void refresh_bbox_upward() {
102 for (auto* node = this; node != nullptr; node = node->parent) {
103 node->refresh_bbox();
104 }

111 void remove_child(RoutingNode<IntPoint>* child_node) {
112 auto iter = std::find(children.begin(), children.end(), child_node);
113 if (iter != children.end()) {
114 children.erase(iter);
115 child_node->parent = nullptr;
116 refresh_bbox_upward();
117 }
